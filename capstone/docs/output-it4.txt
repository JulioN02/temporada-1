# 1. Catalog + stock

  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
  -d '{"name":"Main"}'
curl -s -X POST localhost:3000/api/stock/movements \
  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
  -d '{"type":"adjustment","productId":1,"warehouseId":1,"quantity":10,
       "reason":"initial stock"}'
curl -s localhost:3000/api/stock   # level "10"

# 2. Order + ATOMIC confirm (one tx: sufficiency → order_out → audit → rows)
curl -s -X POST localhost:3000/api/orders \
  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
  -d '{"customerId":1,"lines":[{"productId":1,"qty":3,"unitPrice":"0.10"}]}'
curl -s -X POST localhost:3000/api/orders/1/confirm \
  -H "Authorization: Bearer $TOKEN" -H 'Idempotency-Key: confirm-1'
#   → 200 {order:{state:"confirmed", total:"0.30"}}
curl -s localhost:3000/api/stock            # level "7" (10 − 3, derived)
curl -s localhost:3000/api/stock/movements  # order_out row, sign −1, reason
                                           #   'order 1 confirmed', newest-first
# replay with the SAME key → 200 original, no new movements
curl -s -X POST localhost:3000/api/orders/1/confirm \
  -H "Authorization: Bearer $TOKEN" -H 'Idempotency-Key: confirm-1'

# 3. Insufficient stock → 409 INSUFFICIENT_STOCK, ZERO side effects
curl -s -X POST localhost:3000/api/orders \
  -H "Authorization: Bearer $TOKEN" -H 'Content-Type: application/json' \
  -d '{"customerId":1,"lines":[{"productId":1,"qty":99,"unitPrice":"0.10"}]}'
curl -s -X POST localhost:3000/api/orders/2/confirm \
  -H "Authorization: Bearer $TOKEN" -H 'Idempotency-Key: confirm-2'
#   → 409 {"error":{"code":"INSUFFICIENT_STOCK",...}} — order stays draft,
#     movements/notifications/audit untouched (proven by tests R-ORD-5)

# 4. Low stock fires on crossing (level 7 → confirm qty 4 → 3 < threshold 5)
curl -s -X POST localhost:3000/api/orders/3/confirm \
  -H "Authorization: Bearer $TOKEN" -H 'Idempotency-Key: confirm-3'
#   → low_stock notification rows for manager+operator (R-STK-7)

# 5. Cancel never reverses stock (R-ORD-6)
